RAPIDLINK FRONTEND: ALL SOURCE FILES IN ONE FILE
=================================================
Each file starts with a line  ===== FILE: path =====  and ends where the next one starts.
To turn this back into a normal project (needs Node.js), save the next 12 lines as split.mjs next to this file and run:  node split.mjs

  import fs from "node:fs"; import path from "node:path";
  const txt = fs.readFileSync("rapidlink-frontend-source.txt", "utf8").replace(/\r\n/g, "\n");
  const parts = txt.split(/^===== FILE: (.+?) =====\n/m); // [intro, path1, body1, path2, body2, ...]
  for (let i = 1; i < parts.length; i += 2) {
    const p = parts[i].trim(), body = parts[i + 1].replace(/\n$/, "") + "\n";
    fs.mkdirSync(path.dirname(p), { recursive: true }); fs.writeFileSync(p, body);
  }
  console.log("wrote", (parts.length - 1) / 2, "files");

Then:  cd frontend ; npm install ; npm run build   (result: frontend/dist/index.html)
Keep frontend/ and web/src/ side by side (the frontend imports the preview engine from ../web/src).

// Turns Vite's output (index.html + assets/*.js + assets/*.css) into ONE self-contained dist/index.html,
// the same shape the old page had, so the C++ server (which serves a single file) needs no change.
import { readFileSync, writeFileSync, rmSync, existsSync } from "node:fs";
import { join, dirname } from "node:path";
import { fileURLToPath } from "node:url";

const dist = join(dirname(fileURLToPath(import.meta.url)), "..", "dist");
let html = readFileSync(join(dist, "index.html"), "utf8");
const read = (p) => readFileSync(join(dist, p.replace(/^\//, "")), "utf8");

html = html.replace(/<link rel="stylesheet"[^>]*href="(\/assets\/[^"]+\.css)"[^>]*>/g, (_, p) => `<style>\n${read(p)}\n</style>`);
html = html.replace(/<script type="module"[^>]*src="(\/assets\/[^"]+\.js)"[^>]*><\/script>/g, (_, p) => {
  const js = read(p).replace(/<\/script/gi, "<\\/script");
  return `<script type="module">\n${js}\n</script>`;
});
if (/src="\/assets\//.test(html) || /href="\/assets\//.test(html)) throw new Error("an asset reference is left in index.html");
writeFileSync(join(dist, "index.html"), html);
if (existsSync(join(dist, "assets"))) rmSync(join(dist, "assets"), { recursive: true, force: true });
console.log(`wrote ${join(dist, "index.html")} (${Math.round(html.length / 1024)} KB)`);

function Topbar() {
  const { app, state, ui } = useApp();
  const inAdmin = ui.view.startsWith("admin") || ui.view === "login";
  const mine = ui.myEmergencyId != null && state.emergencies.has(ui.myEmergencyId);
  const link = (hash: string, text: string, on: boolean) => <a className={`navlink${on ? " on" : ""}`} href={`#${hash}`} aria-current={on ? "page" : undefined}>{text}</a>;
  return (
    <>
      <header className="top">
        <div className="topin">
          <a className="brand" href="#report"><Icon.logo /><span>RapidLink</span></a>
          <nav aria-label="Main">
            {link("report", "Report an emergency", ui.view === "report")}
            {mine ? link("track", "Track my emergency", ui.view === "track") : null}
          </nav>
          <span className="grow" />
          {ui.auth.required && ui.auth.admin ? <button className="btn ghost" onClick={() => void app.logout()}>Sign out</button> : null}
          {inAdmin
            ? <a className="btn ghost" href="#report">Public site</a>
            : <a className="btn" href="#admin">{ui.auth.required && !ui.auth.admin ? "Dispatcher sign-in" : "Dispatcher dashboard"}</a>}
        </div>
      </header>
      <div className="demo-banner" role="note"><b>Demonstration system.</b> Prototype assessment, not a medical device and not a substitute for emergency services. In a real emergency call <b>112</b>.</div>
    </>
  );
}

function Toast() {
  const { app, state, ui } = useApp();
  const toast = state.toast;
  // an arrival / escalation message is shown for a few seconds, then cleared
  useEffect(() => {
    if (!toast) return;
    const t = setTimeout(() => { if (app.state.toast && app.state.toast.id === toast.id) app.model.clearToast(); }, 6000);
    return () => clearTimeout(t);
  }, [toast, app]);
  const text = ui.flash || (toast ? toast.text : "");
  return <div id="toast" aria-live="polite">{text ? <div className={`toast${ui.flash ? " err" : ""}`} role="status">{text}</div> : null}</div>;
}

export function App() {
  const { app, ui } = useApp();
  useEffect(() => { void app.boot(); }, [app]);
  useEffect(() => {
    const v = ui.view;
    document.title = (TITLES[v] ?? "Dispatcher dashboard") + " – RapidLink";
  }, [ui.view]);

  // nothing is drawn until the first answer from the backend is in, so the maps start framed on the real data
  if (!app.ready) return <div id="chrome"><p className="pad muted">Loading&hellip;</p></div>;

  const view = ui.view === "report" ? <ReportView /> : ui.view === "track" ? <TrackView /> : ui.view === "login" ? <LoginView /> : <AdminView />;
  return (
    <>
      <div id="chrome"><Topbar /></div>
      <div id="view">{view}</div>
      <div id="drawer">{ui.view.startsWith("admin") ? <Drawer /> : null}</div>
      <Toast />
    </>
  );
}

import { createContext, useContext, useSyncExternalStore } from "react";
import type { Controller } from "./controller";

export const AppContext = createContext<Controller | null>(null);

export function useController(): Controller {
  const c = useContext(AppContext);
  if (!c) throw new Error("AppContext is missing");
  return c;
}

/** The controller plus the current server state and UI state. The component re-renders whenever either changed. */
export function useApp() {
  const app = useController();
  useSyncExternalStore(app.model.subscribe, app.model.getVersion);
  return { app, state: app.state, ui: app.ui };
}

===== FILE: frontend/src/store.ts =====
/* =============================================================
   store.ts  --  The page's copy of the system state, and the page's own UI state.

   The server mirror (`state`) holds NO business rules. It only mirrors what the backend says: first a snapshot, then every
   event (emergency.updated, ambulance.moved, hospital.beds ...). Triage, queue order, ambulance choice, hospital choice and
   ETA are all computed by the backend.

   React reads it through useApp() (useSyncExternalStore): `version` changes whenever something did. Server events are
   coalesced (at most ~12 redraws a second); the user's own actions redraw at once.
   ============================================================= */
import type {
  Ambulance, Emergency, EStatus, GraphInfo, Hospital, Incident, LatLng, MciResult, Notification, QueueEntry,

} from "./types";

function initialServerState(): ServerState {
  return {
    connected: false, mode: "", time: 0, simSpeed: 10, graph: null,
    hospitals: new Map(), ambulances: new Map(), emergencies: new Map(), incidents: new Map(), queue: [], notifications: [], routes: new Map(), toast: null,
  };
}

export class AppModel {
  readonly state: ServerState = initialServerState();
  readonly ui: UiState = initialUi();
  private version = 0;
  private readonly subs = new Set<() => void>();
  private scheduled = false;
  private noteId = 1;
  private needQueue: (() => void) | null = null;

  // ----- useSyncExternalStore plumbing -----
  subscribe = (fn: () => void): (() => void) => { this.subs.add(fn); return () => { this.subs.delete(fn); }; };
  getVersion = (): number => this.version;
  /** Redraw at once (the user did something). */
  touch(): void { this.version++; this.subs.forEach((f) => f()); }
  /** Redraw soon (server events arrive many times a second). */
  private touchSoon(): void {
    if (this.scheduled) return;
    this.scheduled = true;
    setTimeout(() => { this.scheduled = false; this.touch(); }, 80);
  }

  onNeedQueue(fn: () => void): void { this.needQueue = fn; }

  // ----- applying what the backend says -----
  private applySnapshot(s: Snapshot): void {
    const st = this.state;
    st.time = s.time;
    st.simSpeed = s.simSpeed;
    st.hospitals = new Map(s.hospitals.map((h) => [h.id, h]));
    st.ambulances = new Map(s.ambulances.map((a) => [a.id, a]));
    st.emergencies = new Map(s.emergencies.map((e) => [e.id, e]));
    st.queue = s.queue || [];
  }

  private applyEvent(ev: ServerEvent): void {
    const st = this.state;
    if (typeof ev.t === "number" && ev.t > st.time) st.time = ev.t;
    switch (ev.type) {
      case "snapshot": this.applySnapshot(ev.data); break;
      case "connection": st.connected = !!ev.data.connected; break;
      case "incident.updated": st.incidents.set(ev.data.id, ev.data); break;
      case "hospital.beds": st.hospitals.set(ev.data.id, ev.data); break;
      case "ambulance.status": st.ambulances.set(ev.data.id, ev.data); break;
      case "ambulance.moved": {
        const a = st.ambulances.get(ev.data.id);
        if (a) st.ambulances.set(ev.data.id, { ...a, ...ev.data });
        break;
      }
      case "emergency.updated": {
        const old = st.emergencies.get(ev.data.id);
        st.emergencies.set(ev.data.id, ev.data);
        if ((!old || old.status !== ev.data.status) && this.needQueue) this.needQueue();
        break;
      }
      case "emergency.status":
        if ((ev.data.from === "NEW" || ev.data.to === "QUEUED" || ev.data.from === "QUEUED") && this.needQueue) this.needQueue();
        break;
      case "route.planned": st.routes.set(ev.data.emergencyId + ":" + ev.data.leg, ev.data); break;
      case "notification": {
        const d = ev.data;
        const n: Notification = { id: this.noteId++, t: ev.t, kind: d.kind, text: d.text, emergencyId: d.emergencyId, hospitalId: d.hospitalId };
        st.notifications.unshift(n);
        if (st.notifications.length > 300) st.notifications.length = 300;
        if (["arrival", "escalation", "hospital_alert"].includes(d.kind)) st.toast = n;
        break;
      }
      default: break;
    }
  }

  apply(ev: ServerEvent): void { this.applyEvent(ev); this.touchSoon(); }
  setMode(m: "http" | "local"): void { this.state.mode = m; }

export interface Backend {
  readonly mode: "http" | "local";
  request<T = any>(method: "GET" | "POST", path: string, body?: unknown): Promise<ApiResult<T>>;
  onEvent(fn: (e: ServerEvent) => void): void;

interface HttpOptions { base?: string; wsUrl?: string | null; pollMs?: number }

export class HttpBackend implements Backend {
  readonly mode = "http" as const;
  private readonly opts: Required<HttpOptions>;
  private readonly listeners: Array<(e: ServerEvent) => void> = [];
  private ws: WebSocket | null = null;
  private stopped = false;
  private retry = 0;
  private pollTimer: ReturnType<typeof setInterval> | null = null;
  private retryTimer: ReturnType<typeof setTimeout> | null = null;
  private readonly auth = { bearer: "", track: "" };

  constructor(opts: HttpOptions = {}) {
    this.opts = { base: "", wsUrl: null, pollMs: 2000, ...opts };
  }

  private emit(e: ServerEvent): void {
    for (const fn of this.listeners) {
      try { fn(e); } catch (err) { console.error(err); }
    }
  }
  private headers(hasBody: boolean): Record<string, string> {
    const h: Record<string, string> = {};
    if (hasBody) h["Content-Type"] = "application/json";
    if (this.auth.bearer) h["Authorization"] = "Bearer " + this.auth.bearer;
    if (this.auth.track) h["X-Track-Token"] = this.auth.track;
    return h;
  }
  private sendAuth(): void {
    if (this.ws && this.ws.readyState === WebSocket.OPEN) {
      try { this.ws.send(JSON.stringify({ type: "auth", bearer: this.auth.bearer, track: this.auth.track })); } catch { /* the socket is closing */ }
    }
  }
  private wsUrl(): string {
    if (this.opts.wsUrl) return this.opts.wsUrl;
    return (location.protocol === "https:" ? "wss:" : "ws:") + "//" + location.host + "/ws";
  }

  async request<T = any>(method: "GET" | "POST", path: string, body?: unknown): Promise<ApiResult<T>> {
    try {
      const res = await fetch(this.opts.base + path, {
        method, headers: this.headers(body !== undefined),
        body: body === undefined ? undefined : JSON.stringify(body),
      });
      let data: any;
      try { data = await res.json(); } catch { data = { error: "bad_response", message: "The server sent an unreadable answer." }; }
      return { status: res.status, data };
    } catch {
      return { status: 0, data: { error: "network", message: "Cannot reach the server." } as any };
    }
  }

  private startPolling(): void {
    if (this.pollTimer) return;
    this.pollTimer = setInterval(async () => {
      const r = await this.request("GET", "/api/snapshot");
      if (r.status === 200) this.emit({ type: "snapshot", id: 0, t: r.data.time, data: r.data });
    }, this.opts.pollMs);
  }
  private stopPolling(): void {
    if (this.pollTimer) clearInterval(this.pollTimer);
    this.pollTimer = null;
  }

  private connect(): void {
    if (this.stopped) return;
    let ws: WebSocket;
    try { ws = new WebSocket(this.wsUrl()); } catch { this.scheduleRetry(); return; }
    this.ws = ws;
    ws.onopen = () => {
      this.retry = 0;
      this.stopPolling();
      this.sendAuth();                                   // the server answers with a snapshot limited to what these tokens allow
      this.emit({ type: "connection", id: 0, t: 0, data: { connected: true } });
      // The server subscribes us after it sends its first snapshot; re-read once now that we are subscribed, so an event
      // that fell into that gap (for example a status change) cannot stay lost.
      void this.request("GET", "/api/snapshot").then((r) => {
        if (r.status === 200 && !this.stopped) this.emit({ type: "snapshot", id: 0, t: r.data.time, data: r.data });
      });
    };
    ws.onmessage = (m) => {
      let frame: ServerEvent;
      try { frame = JSON.parse(m.data as string); } catch { return; }
      this.emit(frame);
    };
    ws.onclose = () => {
      this.emit({ type: "connection", id: 0, t: 0, data: { connected: false } });
      this.startPolling();
      this.scheduleRetry();
    };
    ws.onerror = () => { try { ws.close(); } catch { /* ignore */ } };
  }
  private scheduleRetry(): void {
    if (this.stopped || this.retryTimer) return;
    const wait = Math.min(10000, 500 * 2 ** this.retry++);
    this.retryTimer = setTimeout(() => { this.retryTimer = null; this.connect(); }, wait);
  }

  onEvent(fn: (e: ServerEvent) => void): void { this.listeners.push(fn); }
  start(): void { this.stopped = false; this.connect(); }
  stop(): void {
    this.stopped = true;
    this.stopPolling();
    if (this.retryTimer) clearTimeout(this.retryTimer);
    if (this.ws) try { this.ws.close(); } catch { /* ignore */ }
  }
  setAuth(patch: { bearer?: string; track?: string }): void { Object.assign(this.auth, patch); this.sendAuth(); }
}

/* =============================================================
   MapView.tsx  --  Leaflet + OpenStreetMap view of the system.

   The component draws what the backend says: every position, status, route and ETA comes from the server state. Nothing is
   decided here. All positions are real latitude/longitude, so real GPS can replace the simulator later without changing
   this file. Tiles come from the public OpenStreetMap server (see its tile usage policy).

   The Leaflet map is created once when the component mounts and updated in place on every render (markers move, routes
   are redrawn only when they change), so zoom and pan survive every update.
   ============================================================= */

import L from "leaflet";

const esc = (s: string) => s.replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c] as string));
const PRI_LABEL = { RED: "Critical", YELLOW: "Urgent", GREEN: "Non-critical" } as const;
const TILES = "https://tile.openstreetmap.org/{z}/{x}/{y}.png";
const ATTRIBUTION = '&copy; <a href="https://www.openstreetmap.org/copyright" target="_blank" rel="noopener">OpenStreetMap</a> contributors';
const FALLBACK = { minLat: 30.26, maxLat: 30.38, minLon: 77.96, maxLon: 78.1 };   // Dehradun, only used before the backend answers
const TO_HOSPITAL_STATES = ["HOSPITAL_SELECTED", "TRANSPORTING", "AT_HOSPITAL", "COMPLETED"];
const FINISHED = ["COMPLETED", "CANCELLED"];

export type MapMode = "picker" | "public" | "admin";

interface MarkerSpec { t: "m"; ll: LatLng; cls: string; html: string; size: [number, number]; anchor: [number, number]; z: number; title: string }

type Spec = MarkerSpec | LineSpec;
interface Item { layer: L.Marker | L.Polyline; sig: string }

export interface MapInstance {

  el: HTMLElement;

  auto: boolean;

}

/** Every live map, by element id (e.g. "map-track"). The tests and RL.map.debug() look here. */
export const mapRegistry = new Map<string, MapInstance>();

/** What is on the map right now (diagnostics and tests). */
export function mapDebug(id: string) {
  const inst = mapRegistry.get("map-" + id);
  if (!inst) return null;
  const markers: Record<string, [number, number]> = {};
  const lines: string[] = [];
  inst.items.forEach((v, k) => {
    if (v.layer instanceof L.Marker) { const p = v.layer.getLatLng(); markers[k] = [p.lat, p.lng]; } else lines.push(k);
  });
  return { zoom: inst.map.getZoom(), center: inst.map.getCenter(), markers, lines };
}

/* ---------- what to draw, from state ---------- */
function desired(state: ServerState, ui: UiState, inst: MapInstance): { want: Map<string, Spec>; em: Emergency | null } {
  const mode = inst.mode;
  const em = inst.eid != null ? state.emergencies.get(inst.eid) ?? null : null;
  const toHospital = !!(em && TO_HOSPITAL_STATES.includes(em.status) && em.hospitalId != null);
  const want = new Map<string, Spec>();
  const marker = (key: string, spec: Omit<MarkerSpec, "t">) => want.set(key, { t: "m", ...spec });
  const line = (key: string, pts: LatLng[], cls: string, pane: string) => want.set(key, { t: "l", pts, cls, pane, sig: cls + "|" + pts.length + "|" + pts[0] + "|" + pts[pts.length - 1] });

  // routes of the focused emergency (public) or of every moving emergency (admin)
  const routes = (e: Emergency) => {
    const t = sel.route(state, e.id, "TO_SCENE"), h = sel.route(state, e.id, "TO_HOSPITAL");
    const onHosp = TO_HOSPITAL_STATES.includes(e.status);
    if (t && t.points.length > 1 && onHosp) line(`r${e.id}:done`, t.points, "route done", "routes");              // already driven: faded
    if (t && t.points.length > 1 && !onHosp && e.status !== "CANCELLED") {
      line(`r${e.id}:sc`, t.points, "route case-casing", "routesCase"); line(`r${e.id}:s`, t.points, "route scene " + e.priority, "routes");
    }
    if (h && h.points.length > 1 && e.status !== "CANCELLED") {
      line(`r${e.id}:hc`, h.points, "route case-casing", "routesCase"); line(`r${e.id}:h`, h.points, "route hosp " + e.priority, "routes");
    }
  };
  if (mode === "public" && em) routes(em);
  if (mode === "admin") for (const e of state.emergencies.values()) if (!FINISHED.includes(e.status) && e.status !== "QUEUED") routes(e);

  // hospitals: once one is chosen the others fade (public view)
  for (const h of state.hospitals.values()) {
    const chosen = toHospital && em!.hospitalId === h.id, dim = mode === "public" && toHospital && !chosen;
    marker("h" + h.id, {
      ll: [h.lat, h.lon], cls: `hosp ${h.status}${dim ? " dim" : ""}${chosen ? " chosen" : ""}`, size: [26, 26], anchor: [13, 13], z: chosen ? 900 : 300,
      html: `${SVG_CROSS}<span class="mlabel">${esc(h.name)}</span>`,
      title: `${h.name}: ICU ${h.icuBeds}, general ${h.generalBeds} (simulated beds)${h.status === "STALE" ? ", data stale" : ""}`,
    });
  }

  // patients
  const pins = mode === "public" ? (em ? [em] : []) : mode === "admin" ? [...state.emergencies.values()].filter((e) => !FINISHED.includes(e.status)) : [];
  for (const e of pins)
    marker("e" + e.id, { ll: [e.lat, e.lon], cls: `pin ${e.priority}`, size: [30, 40], anchor: [15, 38], z: 800, html: svgPin(e.id), title: `#${e.id} ${PRI_LABEL[e.priority]} ${e.type}` });

  // the location being chosen in the report form
  if (mode === "picker" && ui.form.lat != null && ui.form.lon != null)
    marker("pick", { ll: [ui.form.lat, ui.form.lon], cls: "pickpin", size: [40, 40], anchor: [20, 20], z: 1000, html: '<svg viewBox="-20 -20 40 40" width="40" height="40" aria-hidden="true"><circle r="14"/><path d="M-20 0h12M8 0h12M0 -20v12M0 8v12"/></svg>', title: "Selected location" });

  // ambulances (the one on the focused emergency is emphasised in the public view)
  if (mode !== "picker") for (const a of state.ambulances.values()) {
    const mine = !!(em && a.id === em.ambulanceId);
    if (mode === "public" && em && !mine) continue;
    marker("a" + a.id, {
      ll: [a.lat, a.lon], cls: `amb ${a.status}${mine ? " mine" : ""}`, size: [30, 30], anchor: [15, 15], z: mine ? 1100 : 600,
      html: `${SVG_CROSS}<span class="mlabel left">A${a.id}</span>`,
      title: `Ambulance ${a.id}: ${a.status}${a.leg ? ", " + a.leg.replace("_", " ").toLowerCase() : ""}`,
    });
  }
  return { want, em };
}

/* ---------- reconcile the Leaflet layers with the wanted set ---------- */
function reconcile(inst: MapInstance, want: Map<string, Spec>): void {
  for (const [key, cur] of inst.items) if (!want.has(key)) { inst.group.removeLayer(cur.layer); inst.items.delete(key); }
  for (const [key, spec] of want) {
    const cur = inst.items.get(key);
    if (spec.t === "m") {
      const sig = spec.cls + "|" + spec.html + "|" + spec.title;
      if (!cur) {
        // on the picker map markers must not swallow the tap that drops the pin
        const mk = L.marker(spec.ll, { icon: iconFor(spec), title: spec.title, zIndexOffset: spec.z, keyboard: false, riseOnHover: false, interactive: inst.mode !== "picker" });
        mk.addTo(inst.group);
        inst.items.set(key, { layer: mk, sig });
      } else {
        const mk = cur.layer as L.Marker;
        mk.setLatLng(spec.ll);
        if (cur.sig !== sig) {
          mk.setIcon(iconFor(spec));
          mk.options.title = spec.title;
          cur.sig = sig;
          const el = mk.getElement();
          if (el) el.title = spec.title;
        }
      }
    } else if (!cur || cur.sig !== spec.sig) {
      if (cur) inst.group.removeLayer(cur.layer);
      const pl = L.polyline(spec.pts, { className: spec.cls, pane: spec.pane, interactive: false, smoothFactor: 1 });
      pl.addTo(inst.group);
      inst.items.set(key, { layer: pl, sig: spec.sig });
    }
  }
}

function boundsFor(state: ServerState): L.LatLngBounds {
  const pts = [...state.hospitals.values(), ...state.ambulances.values()].map((p) => [p.lat, p.lon] as LatLng);
  if (pts.length) return L.latLngBounds(pts);
  const b = state.graph?.bounds ?? FALLBACK;
  return L.latLngBounds([[b.minLat, b.minLon], [b.maxLat, b.maxLon]]);
}

function fit(inst: MapInstance, bounds: L.LatLngBounds, maxZoom = 16): void {
  inst.auto = true;
  try { inst.map.fitBounds(bounds, { padding: [36, 36], maxZoom, animate: false }); } finally { inst.auto = false; }
}

// thin road lines for small (sample) graphs only; a real city graph is far too big and the tiles show the roads
function drawRoads(inst: MapInstance, state: ServerState): void {
  const g = state.graph;
  if (inst.roads || !g || g.omitted || !g.nodes || !g.edges) return;
  inst.roads = true;
  const n = g.nodes;
  for (const [a, b] of g.edges) L.polyline([[n[a].lat, n[a].lon], [n[b].lat, n[b].lon]], { className: "road", interactive: false, pane: "overlayPane" }).addTo(inst.map);
}

function syncMap(inst: MapInstance, state: ServerState, ui: UiState): void {
  const size = inst.el.clientWidth + "x" + inst.el.clientHeight;
  if (size !== inst.size) { inst.size = size; inst.map.invalidateSize({ pan: false }); }
  drawRoads(inst, state);
  const { want, em } = desired(state, ui, inst);
  reconcile(inst, want);
  // the public tracker re-frames when the situation changes (new leg, hospital chosen), unless the user moved the map
  if (inst.mode === "public" && em && !inst.userMoved) {
    const key = em.status + ":" + (sel.route(state, em.id, "TO_SCENE") ? 1 : 0) + (sel.route(state, em.id, "TO_HOSPITAL") ? 1 : 0);
    if (key !== inst.focusKey) {
      inst.focusKey = key;
      const pts = sel.routePoints(state, em);
      const h = em.hospitalId != null ? state.hospitals.get(em.hospitalId) : null;
      if (h) pts.push([h.lat, h.lon]);
      const a = em.ambulanceId != null ? state.ambulances.get(em.ambulanceId) : null;
      if (a) pts.push([a.lat, a.lon]);
      fit(inst, L.latLngBounds(pts));
    }
  }
}

export function MapView({ id, mode, emergencyId = null }: Props) {
  const { app, state, ui } = useApp();
  const elRef = useRef<HTMLDivElement>(null);
  const instRef = useRef<MapInstance | null>(null);
  const elementId = "map-" + id;

  // create the map once, when the component appears; remove it when it goes away
  useEffect(() => {
    const el = elRef.current;
    if (!el) return;
    const map = L.map(el, { zoomControl: true, attributionControl: true, zoomSnap: 0.5, worldCopyJump: false, minZoom: 9, maxZoom: 19 });
    map.createPane("routesCase").style.zIndex = "410";
    map.createPane("routes").style.zIndex = "420";
    const tiles = L.tileLayer(TILES, { maxZoom: 19, attribution: ATTRIBUTION });
    let failed = 0;
    tiles.on("tileerror", () => { if (++failed >= 3) el.classList.add("notiles"); });
    tiles.on("tileload", () => { failed = 0; el.classList.remove("notiles"); });
    tiles.addTo(map);
    const inst: MapInstance = {
      id: elementId, map, el, mode, eid: emergencyId, group: L.layerGroup().addTo(map), items: new Map(),
      userMoved: false, auto: false, focusKey: "", roads: false, size: "",
    };
    map.on("movestart zoomstart", () => { if (!inst.auto) inst.userMoved = true; });
    map.on("zoomend", () => el.classList.toggle("z-low", map.getZoom() < 13));
    if (mode === "picker") map.on("click", (ev: L.LeafletMouseEvent) => app.pickLocation(ev.latlng.lat, ev.latlng.lng));
    fit(inst, boundsFor(app.state), 14);
    el.classList.toggle("z-low", map.getZoom() < 13);
    instRef.current = inst;
    mapRegistry.set(elementId, inst);
    return () => {
      map.remove();
      mapRegistry.delete(elementId);
      instRef.current = null;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [elementId]);

  // when the emergency being followed changes, frame it afresh
  useEffect(() => {
    const inst = instRef.current;
    if (inst) { inst.eid = emergencyId; inst.userMoved = false; inst.focusKey = ""; }
  }, [emergencyId]);

  // after every render: move markers and redraw routes from the new state
  useEffect(() => {
    if (instRef.current) syncMap(instRef.current, state, ui);
  });

  const label = mode === "picker" ? "Map for choosing the patient’s location. Tap to drop a pin." : "Map of Dehradun with hospitals, ambulances and patients";
  return (
    <div className="map-wrap">
      <div ref={elRef} id={elementId} className={`lmap lmap-${mode}`} data-map={id} data-mode={mode} data-eid={emergencyId ?? ""} role="application" aria-label={label} />
    </div>
  );
}

   WHY THIS EXISTS: a published preview page cannot reach your C++ server. This file lets the
   same UI run on its own. It mirrors the C++ rules and produces the SAME JSON shapes, and
   tests/parity.test.mjs checks it against the real C++ engine. The C++ server remains the
   source of truth; when the page is served by it, this file is not used.
   ============================================================= */
(function (RL) {
  "use strict";
  const INF = Infinity;
  const DISCLAIMER =
    "Prototype assessment for demonstration only. Not a medical device and not a substitute for professional emergency assessment. In a real emergency call 112.";
  const PRI = { RED: 0, YELLOW: 1, GREEN: 2 };
  const CAP = { general: 1, cardiac: 2, trauma: 4, neuro: 8, burn: 16, pediatric: 32 };
  const REQ_CAP = { GENERAL: 1, CARDIAC: 2, TRAUMA: 4, NEURO: 8, BURN: 16, PEDIATRIC: 32 };
  const priorityLabel = (p) => (p === "RED" ? "Critical" : p === "YELLOW" ? "Urgent" : "Non-critical");
  const rnd = (v, d) => Number(Number(v).toFixed(d));

  /* ---------- geometry + graph (RoadGraph.h) ---------- */
  const rad = (d) => (d * Math.PI) / 180;
  function haversineKm(a, b) {
    const dLat = rad(b.lat - a.lat), dLon = rad(b.lon - a.lon);
    const h = Math.sin(dLat / 2) ** 2 + Math.cos(rad(a.lat)) * Math.cos(rad(b.lat)) * Math.sin(dLon / 2) ** 2;
    return 2 * 6371 * Math.asin(Math.sqrt(h));
  }

  class Graph {
    constructor() { this.nodes = []; this.adj = []; this.bounds = null; }
    addNode(name, lat, lon) {
      this.nodes.push({ name, lat, lon });
      this.adj.push([]);
      const b = this.bounds;
      if (!b) this.bounds = { minLat: lat, maxLat: lat, minLon: lon, maxLon: lon };
      else { b.minLat = Math.min(b.minLat, lat); b.maxLat = Math.max(b.maxLat, lat); b.minLon = Math.min(b.minLon, lon); b.maxLon = Math.max(b.maxLon, lon); }
      return this.nodes.length - 1;
    }
    addRoad(a, b, kmph) {
      const km = haversineKm(this.nodes[a], this.nodes[b]);
      const minutes = (km / kmph) * 60;
      this.adj[a].push({ to: b, minutes, km });
      this.adj[b].push({ to: a, minutes, km });
    }
    nearestNode(p) {
      let best = -1, bestKm = INF;
      for (let i = 0; i < this.nodes.length; i++) {
        const d = haversineKm(p, this.nodes[i]);
        if (d < bestKm) { bestKm = d; best = i; }
      }
      return best;
    }
    // Dijkstra (same result as the C++ min-heap version; V is tiny so a linear scan is fine here)
    shortestTree(src) {
      const n = this.nodes.length, dist = Array(n).fill(INF), prev = Array(n).fill(-1), done = Array(n).fill(false);
      dist[src] = 0;
      for (let it = 0; it < n; it++) {
        let u = -1;
        for (let i = 0; i < n; i++) if (!done[i] && (u < 0 || dist[i] < dist[u])) u = i;
        if (u < 0 || dist[u] === INF) break;
        done[u] = true;
        for (const e of this.adj[u]) {
          const nd = dist[u] + e.minutes;
          if (nd < dist[e.to]) { dist[e.to] = nd; prev[e.to] = u; }
        }
      }
      return { dist, prev };
    }
    edgeBetween(a, b) {
      let best = null;
      for (const e of this.adj[a]) if (e.to === b && (!best || e.minutes < best.minutes)) best = e;
      return best;
    }
    pathFromTree(t, src, dst) {
      if (dst < 0 || dst >= this.nodes.length || t.dist[dst] === INF) return { found: false, nodes: [], minutes: 0, km: 0 };
      const nodes = [];
      for (let v = dst; v !== -1; v = t.prev[v]) { nodes.push(v); if (v === src) break; }
      nodes.reverse();
      if (nodes[0] !== src) return { found: false, nodes: [], minutes: 0, km: 0 };
      let km = 0;
      for (let i = 1; i < nodes.length; i++) { const e = this.edgeBetween(nodes[i - 1], nodes[i]); if (e) km += e.km; }
      return { found: true, nodes, minutes: t.dist[dst], km };
    }
    shortestPath(a, b) { return this.pathFromTree(this.shortestTree(a), a, b); }
  }

    const mk = (id, name, node, caps, gen, icu) => ({

  function classify(v) {

  class Engine {
    constructor(cfg) {

      const city = sampleCity();
      this.graph = city.graph; this.hospitals = city.hospitals; this.ambs = city.ambulances;
      this.trip = new Map(); this.base = new Map();
      for (const a of this.ambs) { this.trip.set(a.id, { route: null, progress: 0, leg: 0, emergencyId: -1 }); this.base.set(a.id, a.node); }
      this.records = new Map(); this.queue = []; this.idem = new Map(); this.silenced = new Set(); this.stale = new Map();

      return p.lat >= b.minLat - m && p.lat <= b.maxLat + m && p.lon >= b.minLon - m && p.lon <= b.maxLon + m;

    drain() { const e = this.events; this.events = []; return e; }
    push(type, id, data) { this.events.push({ type, id, t: rnd(this.now, 2), data }); }
    hospital(id) { return this.hospitals.find((h) => h.id === id); }
    amb(id) { return this.ambs.find((a) => a.id === id); }

    rejectReason(h, e, nowI) {
      if (nowI - h.last > this.cfg.maxDataAgeSec) return "its resource data is stale";
      const need = REQ_CAP[e.type];
      if ((h.caps & need) !== need) return "no " + e.type + " capability";
      const icu = e.priority === "RED";
      if ((icu ? h.icu : h.gen) <= 0) return icu ? "no free ICU bed" : "no free general bed";
      return "";
    }

    /* ----- commands ----- */
    submit(type, loc, place, vitals, idemKey) {
      if (idemKey && this.idem.has(idemKey)) return { id: this.idem.get(idemKey), duplicate: true };
      const id = this.nextId++;
      const tr = classify(vitals);
      const rec = { id, type, loc, place, vitals, priority: tr.priority, reason: tr.reason, status: "QUEUED",
        ambulanceId: -1, hospitalId: -1, sceneNode: this.graph.nearestNode(loc), pickedUp: false, bedReserved: false,
        bedIcu: false, bedReleased: false, waitingNoted: false, ambToSceneMin: 0, sceneToHospMin: 0, dwellLeft: 0,
        reportedAt: this.now, completedAt: -1, explanation: "", considered: [], history: [] };
      this.records.set(id, rec);
      if (idemKey) this.idem.set(idemKey, id);
      this.queue.push(id);
      const note = "triaged " + rec.priority + ": " + rec.reason;
      rec.history.push({ t: rnd(this.now, 1), status: "QUEUED", note });
      this.emitEmergency(rec);
      this.push("emergency.status", id, { id, from: "NEW", to: "QUEUED", note });
      this.dispatchPending();
      return { id, duplicate: false };
    }

    tick(dt) {
      if (dt < 0) dt = 0;
      this.now += dt;
      const nowI = Math.trunc(this.now);
      if (this.cfg.autoHeartbeat) for (const h of this.hospitals) if (!this.silenced.has(h.id)) h.last = nowI;
      // a hospital whose data just went stale (or came back) is a state change: announce it once, when it flips
      for (const h of this.hospitals) {
        const stale = nowI - h.last > this.cfg.maxDataAgeSec;
        if (stale !== !!this.stale.get(h.id)) this.emitHospital(h, stale ? "data went stale: no update for over " + this.cfg.maxDataAgeSec + " s" : "data is fresh again");
      }
      for (const r of this.records.values()) {
        if (r.status === "AT_SCENE" && !r.pickedUp) { r.dwellLeft -= dt; if (r.dwellLeft <= 0) this.pickup(r); }
        else if (r.status === "NO_HOSPITAL") this.selectHospital(r, true);
        else if (r.status === "COMPLETED" && this.cfg.autoDischargeSec > 0 && r.bedReserved && !r.bedReleased &&
                 this.now - r.completedAt >= this.cfg.autoDischargeSec) {
          this.releaseBed(r, "patient discharged (length of stay elapsed)");
          this.emitEmergency(r);
        }
      }
      for (const [aid, st] of this.trip) if (st.leg !== 0) this.advance(aid, st, dt);
      this.dispatchPending();
    }

    cancel(id) {
      const r = this.records.get(id);
      if (!r) return "unknown emergency";
      if (!E_NEXT[r.status].includes("CANCELLED")) return "cannot cancel an emergency in status " + r.status;
      this.releaseBed(r, "reservation released: emergency cancelled");
      if (r.status === "QUEUED") this.queue = this.queue.filter((x) => x !== id);
      else if (r.ambulanceId >= 0) {
        const a = this.amb(r.ambulanceId), st = this.trip.get(a.id), from = a.node;
        this.setAmb(a, "RETURNING"); st.emergencyId = -1; st.leg = 0;
        this.startReturn(a, st, from);
      }
      this.transition(r, "CANCELLED", "cancelled by operator");
      this.emitEmergency(r);
      this.dispatchPending();
      return "";
    }

    discharge(id) {
      const r = this.records.get(id);
      if (!r) return "unknown emergency";
      if (r.status !== "COMPLETED") return "only COMPLETED emergencies can be discharged";
      if (!r.bedReserved || r.bedReleased) return "bed already released";
      this.releaseBed(r, "patient discharged");
      this.emitEmergency(r);
      return "";
    }

    setMaintenance(id, on) {
      const a = this.amb(id);
      if (!a) return "unknown ambulance";
      if (on) {
        if (a.status !== "AVAILABLE") return "only AVAILABLE ambulances can go to maintenance";
        this.setAmb(a, "MAINTENANCE");
      } else {
        if (a.status !== "MAINTENANCE") return "ambulance is not in maintenance";
        this.setAmb(a, "AVAILABLE"); this.dispatchPending();
      }
      return "";
    }

    updateBeds(id, gen, icu) {
      const h = this.hospital(id);
      if (!h) return "unknown hospital";
      if (gen < 0 || icu < 0 || gen > 10000 || icu > 10000) return "bed counts must be 0..10000";
      h.gen = gen; h.icu = icu; h.last = Math.trunc(this.now);
      this.emitHospital(h, "staff update"); this.dispatchPending();
      return "";
    }
    heartbeat(id) {
      const h = this.hospital(id);
      if (!h) return "unknown hospital";
      h.last = Math.trunc(this.now); this.emitHospital(h, "heartbeat");
      return "";
    }
    setHeartbeatEnabled(id, enabled) {
      const h = this.hospital(id);
      if (!h) return "unknown hospital";
      if (enabled) this.silenced.delete(id); else this.silenced.add(id);
      this.emitHospital(h, enabled ? "heartbeat resumed" : "heartbeat stopped (simulating an outage)");
      return "";
    }
    positionUpdate(id, loc) {
      const a = this.amb(id);
      if (!a) return "unknown ambulance";
      if (a.status !== "AVAILABLE") return "ambulance is not AVAILABLE; positions of active units are controlled by the simulator";
      if (!this.inServiceArea(loc)) return "position is outside the service area";
      a.lat = loc.lat; a.lon = loc.lon; a.node = this.graph.nearestNode(loc);
      this.base.set(id, a.node);
      this.emitMoved(a, this.trip.get(id)); this.emitAmbulance(a); this.dispatchPending();
      return "";
    }

    /* ----- stage 1: ambulance allocation ----- */
    sortedQueue() {
      return this.queue.slice().sort((x, y) => PRI[this.records.get(x).priority] - PRI[this.records.get(y).priority] || x - y);
    }
    dispatchPending() {
      while (this.queue.length) {
        const id = this.sortedQueue()[0];
        if (!this.allocateAmbulance(this.records.get(id))) break;
        this.queue = this.queue.filter((x) => x !== id);
      }
    }
    pickNearestAvailable(dist) {
      let best = null, bestT = INF;
      for (const a of this.ambs) {
        if (a.status !== "AVAILABLE") continue;
        const t = dist[a.node];
        if (t === INF) continue;
        const clearlyCloser = t < bestT - 0.01, tied = Math.abs(t - bestT) <= 0.01;
        if (!best || clearlyCloser || (tied && a.idleSince < best.idleSince)) { best = a; bestT = t; }
      }
      return best;
    }
    allocateAmbulance(r) {
      const tree = this.graph.shortestTree(r.sceneNode);
      const amb = this.pickNearestAvailable(tree.dist);
      if (!amb) {
        if (!r.waitingNoted) {
          r.waitingNoted = true;
          this.notify("dispatch", "No ambulance is available for emergency #" + r.id + "; the case stays in the queue and will be served in priority order.", r.id, -1);
        }
        return false;
      }
      const p = this.graph.pathFromTree(tree, r.sceneNode, amb.node);
      if (!p.found) return false;
      p.nodes.reverse();
      const st = this.trip.get(amb.id);
      st.route = this.makeRoute(p.nodes); st.progress = 0; st.leg = 1; st.emergencyId = r.id;
      r.ambulanceId = amb.id; r.ambToSceneMin = tree.dist[amb.node];
      this.setAmb(amb, "DISPATCHED");
      this.transition(r, "EN_ROUTE_TO_SCENE", "ambulance A" + amb.id + " dispatched");
      this.emitRoute(r.id, "TO_SCENE", st.route);
      this.notify("dispatch", "Ambulance A" + amb.id + " dispatched to emergency #" + r.id + " (" + st.route.totalMin.toFixed(1) + " min away, " + st.route.totalKm.toFixed(1) + " km).", r.id, -1);
      this.emitEmergency(r);
      return true;
    }

    /* ----- movement ----- */
    advance(aid, st, dt) {
      const a = this.amb(aid);
      st.progress += (dt / 60) * this.cfg.simSpeed;
      const done = st.progress >= st.route.totalMin - 1e-9;
      if (done) st.progress = st.route.totalMin;
      const p = this.positionOn(st.route, st.progress);
      a.lat = p.lat; a.lon = p.lon; a.node = p.node;
      this.emitMoved(a, st);
      if (done) this.arrive(a, st);
    }
    arrive(a, st) {
      if (st.leg === 1) {
        const r = this.records.get(st.emergencyId);
        st.leg = 0;
        this.transition(r, "AT_SCENE", "ambulance arrived at the scene");
        r.dwellLeft = this.cfg.sceneDwellSec;
        this.notify("arrival", "Ambulance A" + a.id + " has arrived at the patient (emergency #" + r.id + ").", r.id, -1);
        this.emitEmergency(r);
      } else if (st.leg === 2) {
        const r = this.records.get(st.emergencyId), hname = this.hospital(r.hospitalId).name;
        st.leg = 0;
        this.transition(r, "AT_HOSPITAL", "arrived at " + hname);
        this.transition(r, "COMPLETED", "handover complete; ambulance returning to base");
        r.completedAt = this.now;
        this.notify("arrival", "Ambulance A" + a.id + " arrived at " + hname + ". Emergency #" + r.id + " completed.", r.id, r.hospitalId);
        this.setAmb(a, "RETURNING");
        const from = st.route.nodes[st.route.nodes.length - 1];
        st.emergencyId = -1;
        this.startReturn(a, st, from);
        this.emitEmergency(r);
      } else if (st.leg === 3) {
        st.leg = 0;
        this.setAmb(a, "AVAILABLE");
      }
    }
    startReturn(a, st, fromNode) {
      const home = this.base.get(a.id);
      const goHome = () => { const n = this.graph.nodes[home]; a.lat = n.lat; a.lon = n.lon; a.node = home; st.leg = 0; this.setAmb(a, "AVAILABLE"); };
      if (fromNode === home) return goHome();
      const p = this.graph.shortestPath(fromNode, home);
      if (!p.found) return goHome();
      st.route = this.makeRoute(p.nodes); st.progress = 0; st.leg = 3;
    }

    /* ----- stage 2: hospital selection ----- */
    pickup(r) {
      const a = this.amb(r.ambulanceId);
      this.setAmb(a, "BUSY");
      r.pickedUp = true;
      this.notify("pickup", "Patient picked up by ambulance A" + a.id + " (emergency #" + r.id + ").", r.id, -1);
      this.selectHospital(r, false);
    }
    selectHospital(r, quiet) {
      const icu = r.priority === "RED", nowI = Math.trunc(this.now);
      const tree = this.graph.shortestTree(r.sceneNode);
      const free = (h) => (icu ? h.icu : h.gen);
      const better = (x, y) => {
        const tx = tree.dist[this.hospitals[x].node], ty = tree.dist[this.hospitals[y].node];
        if (tx < ty - 0.01) return true;
        if (tx > ty + 0.01) return false;
        return free(this.hospitals[x]) > free(this.hospitals[y]);
      };
      let best = -1, second = -1, nearestAny = -1;
      const cons = [];
      this.hospitals.forEach((h, i) => {
        const t = tree.dist[h.node], reachable = t !== INF;
        let reason = "";
        if (!reachable) reason = "not reachable by road";
        else {
          reason = this.rejectReason(h, r, nowI);
          if (nearestAny < 0 || t < tree.dist[this.hospitals[nearestAny].node]) nearestAny = i;
        }
        cons.push({ hospitalId: h.id, name: h.name, minutes: reachable ? rnd(t, 1) : -1, accepted: reachable && reason === "", reason, chosen: false });
        if (!reachable || reason) return;
        if (best < 0 || better(i, best)) { second = best; best = i; }
        else if (second < 0 || better(i, second)) second = i;
      });
      r.considered = cons;
      if (best < 0) {
        r.explanation = "No reachable hospital has the capability, capacity and fresh data for this " + r.type +
          " case. Escalate to a human dispatcher. The ambulance stays with the patient and the search is retried automatically.";
        if (r.status !== "NO_HOSPITAL") {
          this.transition(r, "NO_HOSPITAL", "no qualified hospital");
          this.notify("escalation", "Emergency #" + r.id + ": no qualified hospital available. Human dispatcher needed.", r.id, -1);
          this.emitEmergency(r);
        } else if (!quiet) this.emitEmergency(r);
        return false;
      }
      const ch = this.hospitals[best];
      if (icu) ch.icu--; else ch.gen--;
      r.bedReserved = true; r.bedIcu = icu; r.bedReleased = false; r.hospitalId = ch.id;
      const toH = tree.dist[ch.node];
      r.sceneToHospMin = toH;
      cons[best].chosen = true;
      let why = r.priority + " " + r.type + " case: assigned to " + ch.name + " (" + toH.toFixed(1) + " min from scene), the nearest hospital with " +
        r.type + " capability and a free " + (icu ? "ICU" : "general") + " bed.";
      if (nearestAny >= 0 && nearestAny !== best) {
        const n = this.hospitals[nearestAny];
        why += " Skipped nearest hospital " + n.name + " (" + tree.dist[n.node].toFixed(1) + " min): " + this.rejectReason(n, r, nowI) + ".";
      }
      if (second >= 0) why += " Next-best option: " + this.hospitals[second].name + " (" + tree.dist[this.hospitals[second].node].toFixed(1) + " min).";
      if (toH > this.cfg.preferredRadiusMin) why += " Note: no qualified hospital within " + this.cfg.preferredRadiusMin.toFixed(1) + " min, so the search was widened.";
      why += " Ambulance A" + r.ambulanceId + " was the nearest available unit (" + r.ambToSceneMin.toFixed(1) + " min from its position at dispatch).";
      r.explanation = why;
      const p = this.graph.pathFromTree(tree, r.sceneNode, ch.node);
      const st = this.trip.get(r.ambulanceId);
      st.route = this.makeRoute(p.nodes); st.progress = 0; st.leg = 2;
      this.transition(r, "HOSPITAL_SELECTED", "hospital " + ch.name + " selected, bed reserved");
      this.transition(r, "TRANSPORTING", "transporting to " + ch.name);
      this.emitHospital(ch, "bed reserved for emergency #" + r.id);
      this.emitRoute(r.id, "TO_HOSPITAL", st.route);
      this.notify("hospital_alert", "[ALERT -> " + ch.name + "] " + r.priority + " " + r.type + " emergency #" + r.id + " at " + r.place + ". Triage: " + r.reason +
        ". Ambulance A" + r.ambulanceId + " (ETA to hospital " + r.sceneToHospMin.toFixed(1) + " min). Prepare " + (icu ? "ICU bed" : "general bed") + ".", r.id, ch.id);
      this.emitEmergency(r);
      return true;
    }

    /* ----- internals ----- */
    transition(r, to, note) {
      if (!E_NEXT[r.status].includes(to)) throw new Error("illegal emergency transition " + r.status + " -> " + to);
      const from = r.status;
      r.status = to;
      r.history.push({ t: rnd(this.now, 1), status: to, note });
      this.push("emergency.status", r.id, { id: r.id, from, to, note });
    }
    setAmb(a, to) {
      if (!A_NEXT[a.status].includes(to)) throw new Error("illegal ambulance transition " + a.status + " -> " + to);
      a.status = to;
      if (to === "AVAILABLE") a.idleSince = Math.trunc(this.now);
      this.emitAmbulance(a);
    }
    releaseBed(r, reason) {
      if (!r.bedReserved || r.bedReleased) return;
      const h = this.hospital(r.hospitalId);
      if (r.bedIcu) h.icu++; else h.gen++;
      r.bedReleased = true;
      this.emitHospital(h, reason + " (emergency #" + r.id + ")");
    }
    makeRoute(nodes) {
      const R = { nodes: nodes.slice(), pts: [], cumMin: [], cumKm: [], totalMin: 0, totalKm: 0 };
      let m = 0, k = 0;
      nodes.forEach((n, i) => {
        if (i > 0) { const e = this.graph.edgeBetween(nodes[i - 1], n); if (e) { m += e.minutes; k += e.km; } }
        R.pts.push({ lat: this.graph.nodes[n].lat, lon: this.graph.nodes[n].lon });
        R.cumMin.push(m); R.cumKm.push(k);
      });
      R.totalMin = m; R.totalKm = k;
      return R;
    }
    positionOn(R, m) {
      const last = R.pts.length - 1;
      if (R.pts.length === 1 || R.totalMin <= 0) return { lat: R.pts[last].lat, lon: R.pts[last].lon, node: R.nodes[last], km: R.totalKm, speed: 0 };
      let k = 0;
      while (k < R.cumMin.length && R.cumMin[k] <= m) k++;            // first index with cumMin > m
      if (k === 0) k = 1;
      if (k >= R.cumMin.length) k = R.cumMin.length - 1;
      const i = k - 1, span = R.cumMin[k] - R.cumMin[i];
      let f = span > 0 ? (m - R.cumMin[i]) / span : 1;
      f = Math.max(0, Math.min(1, f));
      return { lat: R.pts[i].lat + f * (R.pts[k].lat - R.pts[i].lat), lon: R.pts[i].lon + f * (R.pts[k].lon - R.pts[i].lon),
        node: f >= 1 ? R.nodes[k] : R.nodes[i], km: R.cumKm[i] + f * (R.cumKm[k] - R.cumKm[i]),
        speed: span > 0 ? ((R.cumKm[k] - R.cumKm[i]) / span) * 60 : 0 };
    }
    progressOf(st) {
      if (st.leg === 0 || st.route.totalMin <= 0) return { etaSec: 0, remKm: 0, frac: 1, speed: 0 };
      const p = this.positionOn(st.route, st.progress);
      return { etaSec: (Math.max(0, st.route.totalMin - st.progress) * 60) / this.cfg.simSpeed, remKm: Math.max(0, st.route.totalKm - p.km),
        frac: st.progress / st.route.totalMin, speed: p.speed };
    }

    /* ----- JSON (same shapes as the C++ API) ----- */
    destinationOf(st, aid) {
      if (st.leg === 1 && this.records.has(st.emergencyId)) return this.records.get(st.emergencyId).place;
      if (st.leg === 2 && this.records.has(st.emergencyId)) return this.hospital(this.records.get(st.emergencyId).hospitalId).name;
      if (st.leg === 3) return this.graph.nodes[this.base.get(aid)].name;
      return null;
    }
    ambulanceJson(a) {
      const st = this.trip.get(a.id), pg = this.progressOf(st), dest = this.destinationOf(st, a.id);
      return { id: a.id, status: a.status, lat: rnd(a.lat, 6), lon: rnd(a.lon, 6), location: this.graph.nodes[a.node].name,
        base: this.graph.nodes[this.base.get(a.id)].name, emergencyId: st.emergencyId >= 0 ? st.emergencyId : null,
        leg: st.leg ? LEG[st.leg] : null, destination: dest, etaSec: rnd(pg.etaSec, 1), remainingKm: rnd(pg.remKm, 2),
        progress: rnd(pg.frac, 3), speedKmh: rnd(pg.speed, 1), legKm: st.leg ? rnd(st.route.totalKm, 2) : 0, legMin: st.leg ? rnd(st.route.totalMin, 2) : 0 };
    }
    hospitalJson(h) {
      const age = Math.trunc(this.now) - h.last, stale = age > this.cfg.maxDataAgeSec;
      const caps = Object.keys(CAP).filter((k) => h.caps & CAP[k]);
      return { id: h.id, name: h.name, lat: rnd(this.graph.nodes[h.node].lat, 6), lon: rnd(this.graph.nodes[h.node].lon, 6),
        location: this.graph.nodes[h.node].name, capabilities: caps, generalBeds: h.gen, icuBeds: h.icu, dataAgeSec: age,
        status: stale ? "STALE" : "ONLINE", heartbeatEnabled: !this.silenced.has(h.id), updatedAt: h.last };
    }
    recordJson(r, withRoute) {
      const o = { id: r.id, status: r.status, priority: r.priority, priorityLabel: priorityLabel(r.priority), triageReason: r.reason,
        disclaimer: DISCLAIMER, type: r.type, place: r.place, lat: rnd(r.loc.lat, 6), lon: rnd(r.loc.lon, 6),
        sceneJunction: this.graph.nodes[r.sceneNode].name, reportedAt: rnd(r.reportedAt, 1),
        ambulanceId: r.ambulanceId >= 0 ? r.ambulanceId : null };
      if (r.hospitalId >= 0) { o.hospitalId = r.hospitalId; o.hospitalName = this.hospital(r.hospitalId).name; o.bedType = r.bedIcu ? "ICU" : "GENERAL"; o.bedReserved = r.bedReserved && !r.bedReleased; }
      else { o.hospitalId = null; o.hospitalName = null; }
      o.ambulanceToSceneMin = rnd(r.ambToSceneMin, 1); o.sceneToHospitalMin = rnd(r.sceneToHospMin, 1);
      o.explanation = r.explanation; o.considered = r.considered.map((c) => Object.assign({}, c));
      o.history = r.history.map((h) => Object.assign({}, h));
      if (r.ambulanceId >= 0 && holdsAmbulance(r.status)) {
        const st = this.trip.get(r.ambulanceId);
        if (st.leg === 1 || st.leg === 2) {
          const pg = this.progressOf(st);
          o.leg = LEG[st.leg]; o.etaSec = rnd(pg.etaSec, 1); o.remainingKm = rnd(pg.remKm, 2); o.progress = rnd(pg.frac, 3);
          o.legKm = rnd(st.route.totalKm, 2); o.legMin = rnd(st.route.totalMin, 2);
          if (withRoute) { o.route = st.route.pts.map((p) => [rnd(p.lat, 6), rnd(p.lon, 6)]); o.routeJunctions = st.route.nodes.map((n) => this.graph.nodes[n].name); }
        }
      }
      return o;
    }
    emergencyJson(id) { const r = this.records.get(id); return r ? this.recordJson(r, true) : null; }
    emergenciesJson() { return [...this.records.values()].map((r) => this.recordJson(r, false)); }
    queueJson() {
      return this.sortedQueue().map((id, i) => { const r = this.records.get(id);
        return { position: i + 1, id, priority: r.priority, type: r.type, place: r.place, reason: r.reason, waitedSec: rnd(this.now - r.reportedAt, 1) }; });
    }
    ambulancesJson() { return this.ambs.map((a) => this.ambulanceJson(a)); }
    hospitalsJson() { return this.hospitals.map((h) => this.hospitalJson(h)); }
    snapshotJson() {
      return { time: rnd(this.now, 1), simSpeed: this.cfg.simSpeed, hospitals: this.hospitalsJson(), ambulances: this.ambulancesJson(),
        emergencies: this.emergenciesJson(), queue: this.queueJson() };
    }
    graphJson() {
      const g = this.graph, edges = [];
      g.adj.forEach((list, a) => list.forEach((e) => { if (a < e.to) edges.push([a, e.to, rnd(e.km, 3)]); }));
      return { nodeCount: g.nodes.length, bounds: Object.assign({}, g.bounds), omitted: false,
        nodes: g.nodes.map((n, i) => ({ id: i, name: n.name, lat: n.lat, lon: n.lon })), edges };
    }

    /* ----- events ----- */
    emitEmergency(r) { this.push("emergency.updated", r.id, this.recordJson(r, false)); }
    emitAmbulance(a) { this.push("ambulance.status", a.id, this.ambulanceJson(a)); }
    emitMoved(a, st) {
      const pg = this.progressOf(st);
      this.push("ambulance.moved", a.id, { id: a.id, status: a.status, lat: rnd(a.lat, 6), lon: rnd(a.lon, 6), location: this.graph.nodes[a.node].name, leg: st.leg ? LEG[st.leg] : null,
        emergencyId: st.emergencyId >= 0 ? st.emergencyId : null, destination: this.destinationOf(st, a.id), etaSec: rnd(pg.etaSec, 1), remainingKm: rnd(pg.remKm, 2), progress: rnd(pg.frac, 3),
        speedKmh: rnd(pg.speed, 1), legKm: st.leg ? rnd(st.route.totalKm, 2) : 0 });
    }
    emitHospital(h, reason) { this.stale.set(h.id, Math.trunc(this.now) - h.last > this.cfg.maxDataAgeSec); this.push("hospital.beds", h.id, Object.assign(this.hospitalJson(h), { reason })); }
    emitRoute(eid, leg, R) {
      this.push("route.planned", eid, { emergencyId: eid, leg, minutes: rnd(R.totalMin, 2), km: rnd(R.totalKm, 2),
        points: R.pts.map((p) => [rnd(p.lat, 6), rnd(p.lon, 6)]), junctions: R.nodes.map((n) => this.graph.nodes[n].name) });
    }
    notify(kind, text, emergencyId, hospitalId) {
      this.push("notification", emergencyId >= 0 ? emergencyId : 0, { kind, text, emergencyId: emergencyId >= 0 ? emergencyId : null, hospitalId: hospitalId >= 0 ? hospitalId : null });
    }
  }

  RL.LocalEngine = Engine;
  RL.classifyVitals = classify;
  RL.sampleCity = sampleCity;
  RL.haversineKm = haversineKm;
  RL.DISCLAIMER = DISCLAIMER;
})((globalThis.RL = globalThis.RL || {}));

/* =============================================================
   mci.js  --  Port of Scenario.h, SelfishRouting.h, MinCostFlow.h, CoordinatedRouting.h, Mci.h

   Uses an exact MT19937 (the generator behind std::mt19937), so for the same seed this file
   draws the same random patients as the C++ code and reproduces its numbers.
   tests/mci.test.mjs checks it against the real C++ engine.
   ============================================================= */
(function (RL) {
  "use strict";
  const INF = Infinity;

  class MT19937 {
    constructor(seed) {
      this.mt = new Uint32Array(624);
      this.i = 624;
      this.mt[0] = seed >>> 0;
      for (let k = 1; k < 624; k++) {
        const prev = this.mt[k - 1] ^ (this.mt[k - 1] >>> 30);
        this.mt[k] = (Math.imul(1812433253, prev) + k) >>> 0;
      }
    }
    next() {
      const mt = this.mt;
      if (this.i >= 624) {
        for (let k = 0; k < 624; k++) {
          const y = (mt[k] & 0x80000000) | (mt[(k + 1) % 624] & 0x7fffffff);
          mt[k] = (mt[(k + 397) % 624] ^ (y >>> 1) ^ (y & 1 ? 0x9908b0df : 0)) >>> 0;
        }
        this.i = 0;
      }
      let y = mt[this.i++];
      y ^= y >>> 11;
      y ^= (y << 7) & 0x9d2c5680;
      y ^= (y << 15) & 0xefc60000;
      y ^= y >>> 18;
      return y >>> 0;
    }
  }
  const randInt = (r, lo, hi) => lo + (r.next() % (hi - lo + 1));
  const rand01 = (r) => (r.next() >>> 8) / 16777216.0;
  const PRI = { RED: 0, YELLOW: 1, GREEN: 2 };
  const CAP = { general: 1, cardiac: 2, trauma: 4, neuro: 8, burn: 16, pediatric: 32 };
  const REQ_CAP = { GENERAL: 1, CARDIAC: 2, TRAUMA: 4, NEURO: 8, BURN: 16, PEDIATRIC: 32 };
  const PENALTY = [1000, 500, 200];

  function generatePatients(g, n, rng) {
    const epi = randInt(rng, 0, g.nodes.length - 1);
    const t = g.shortestTree(epi).dist;
    const order = g.nodes.map((_, i) => i).sort((a, b) => t[a] - t[b]);
    const cluster = Math.min(4, g.nodes.length);
    const out = [];
    for (let k = 0; k < n; k++) {
      const node = order[randInt(rng, 0, cluster - 1)];
      const canWalk = rand01(rng) < 0.45;
      const rr = randInt(rng, 12, 36);
      const pulse = randInt(rng, 60, 130);
      const a = rand01(rng);
      const avpu = a < 0.88 ? "A" : a < 0.95 ? "V" : a < 0.98 ? "P" : "U";
      const bleed = rand01(rng) < 0.08;
      const ty = rand01(rng);
      const type = ty < 0.4 ? "GENERAL" : ty < 0.8 ? "TRAUMA" : ty < 0.9 ? "BURN" : "CARDIAC";
      const tr = RL.classifyVitals({ canWalk, rr, pulse, avpu, bleed });
      out.push({ id: k + 1, node, type, priority: tr.priority, place: g.nodes[node].name });
    }
    return out;
  }

  function buildScenario(g, hospitals, patients, cfg) {
    const pools = [];
    hospitals.forEach((h, hi) => {
      if (h.icu > 0) pools.push({ hi, icu: true, cap: h.icu, label: h.name + " (ICU)" });
      const gen = Math.min(h.gen, cfg.surge);
      if (gen > 0) pools.push({ hi, icu: false, cap: gen, label: h.name + " (general)" });
    });
    const travel = [], valid = [];
    for (const e of patients) {
      const t = g.shortestTree(e.node).dist, tr = [], vl = [];
      pools.forEach((p, q) => {
        const h = hospitals[p.hi];
        tr.push(t[h.node]);
        if (t[h.node] === INF) return;
        if (p.icu !== (e.priority === "RED")) return;
        const need = REQ_CAP[e.type];
        if ((h.caps & need) !== need) return;
        vl.push(q);
      });
      travel.push(tr); valid.push(vl);
    }
    return { cfg, patients, pools, travel, valid };
  }
  const penaltyOf = (s, i) => PENALTY[PRI[s.patients[i].priority]];

  function evaluate(s, poolOf) {
    const m = { total: 0, waitSum: 0, maxWait: 0, served: 0, diverted: 0, divertedRed: 0, redWait: 0, redServed: 0, load: Array(s.pools.length).fill(0) };
    poolOf.forEach((q) => { if (q >= 0) m.load[q]++; });
    poolOf.forEach((q, i) => {
      if (q < 0) { m.total += penaltyOf(s, i); m.diverted++; if (s.patients[i].priority === "RED") m.divertedRed++; return; }
      const w = s.travel[i][q] + s.cfg.congestion * m.load[q];
      m.total += w; m.waitSum += w; m.maxWait = Math.max(m.maxWait, w); m.served++;
      if (s.patients[i].priority === "RED") { m.redWait += w; m.redServed++; }
    });
    m.meanWait = m.served ? m.waitSum / m.served : 0;
    m.meanRed = m.redServed ? m.redWait / m.redServed : 0;
    return m;
  }

  function greedyNearest(s) {
    const P = s.patients.length;
    const order = s.patients.map((_, i) => i).sort((a, b) => PRI[s.patients[a].priority] - PRI[s.patients[b].priority] || a - b);
    const load = Array(s.pools.length).fill(0), poolOf = Array(P).fill(-1);
    for (const i of order) {
      let best = -1;
      for (const q of s.valid[i]) {
        if (load[q] >= s.pools[q].cap) continue;
        if (best < 0 || s.travel[i][q] < s.travel[i][best]) best = q;
      }
      if (best >= 0) { poolOf[i] = best; load[best]++; }
    }
    return poolOf;
  }
  function bestMove(s, i, cur, load) {
    const curCost = cur >= 0 ? s.travel[i][cur] + s.cfg.congestion * load[cur] : penaltyOf(s, i);
    let bq = -2, bc = curCost - 1e-9;
    for (const q of s.valid[i]) {
      if (q === cur || load[q] >= s.pools[q].cap) continue;
      const c = s.travel[i][q] + s.cfg.congestion * (load[q] + 1);
      if (c < bc) { bc = c; bq = q; }
    }
    return bq;
  }
  function bestResponse(s, start) {
    const poolOf = start.slice(), load = Array(s.pools.length).fill(0);
    poolOf.forEach((q) => { if (q >= 0) load[q]++; });
    for (let round = 1; round <= 1000; round++) {
      let moved = false;
      for (let i = 0; i < s.patients.length; i++) {
        const q = bestMove(s, i, poolOf[i], load);
        if (q === -2) continue;
        if (poolOf[i] >= 0) load[poolOf[i]]--;
        poolOf[i] = q; load[q]++; moved = true;
      }
      if (!moved) return { poolOf, rounds: round, converged: true };
    }
    return { poolOf, rounds: 1000, converged: false };
  }

  class MinCostFlow {
    constructor(n) { this.g = Array.from({ length: n }, () => []); this.to = []; this.cap = []; this.cost = []; }
    addEdge(u, v, cap, cost) {
      const id = this.to.length;
      this.to.push(v); this.cap.push(cap); this.cost.push(cost); this.g[u].push(id);
      this.to.push(u); this.cap.push(0); this.cost.push(-cost); this.g[v].push(id + 1);
      return id;
    }
    solve(s, t, maxFlow) {
      const n = this.g.length;
      let flow = 0, cost = 0;
      while (flow < maxFlow) {
        const dist = Array(n).fill(INF), prevE = Array(n).fill(-1), inq = Array(n).fill(false);
        const q = [s]; dist[s] = 0; let head = 0;
        while (head < q.length) {
          const u = q[head++]; inq[u] = false;
          for (const id of this.g[u]) {
            if (this.cap[id] > 0 && dist[u] + this.cost[id] < dist[this.to[id]]) {
              dist[this.to[id]] = dist[u] + this.cost[id]; prevE[this.to[id]] = id;
              if (!inq[this.to[id]]) { inq[this.to[id]] = true; q.push(this.to[id]); }
            }
          }
        }
        if (dist[t] === INF) break;
        let push = maxFlow - flow;
        for (let v = t; v !== s; v = this.to[prevE[v] ^ 1]) push = Math.min(push, this.cap[prevE[v]]);
        for (let v = t; v !== s; v = this.to[prevE[v] ^ 1]) { this.cap[prevE[v]] -= push; this.cap[prevE[v] ^ 1] += push; }
        flow += push; cost += push * dist[t];
      }
      return { flow, cost };
    }
    flowOn(id) { return this.cap[id ^ 1]; }
  }
  function coordinated(s) {
    const P = s.patients.length, Q = s.pools.length, SRC = 0, PAT0 = 1, POOL0 = 1 + P, DIV = 1 + P + Q, SINK = DIV + 1;
    const milli = (x) => Math.round(x * 1000);
    const net = new MinCostFlow(SINK + 1), arcs = [];
    for (let i = 0; i < P; i++) {
      net.addEdge(SRC, PAT0 + i, 1, 0); arcs.push([]);
      for (const q of s.valid[i]) arcs[i].push([net.addEdge(PAT0 + i, POOL0 + q, 1, milli(s.travel[i][q])), q]);
      net.addEdge(PAT0 + i, DIV, 1, milli(penaltyOf(s, i)));
    }
    for (let q = 0; q < Q; q++) {
      const usable = Math.min(s.pools[q].cap, P);
      for (let k = 1; k <= usable; k++) net.addEdge(POOL0 + q, SINK, 1, milli(s.cfg.congestion * (2 * k - 1)));
    }
    net.addEdge(DIV, SINK, P, 0);
    net.solve(SRC, SINK, P);
    const poolOf = Array(P).fill(-1);
    for (let i = 0; i < P; i++) for (const [e, q] of arcs[i]) if (net.flowOn(e) === 1) poolOf[i] = q;
    return poolOf;
  }

  function runOne(g, hospitals, rng, p) {
    const s = buildScenario(g, hospitals, generatePatients(g, p.patients, rng), { congestion: p.congestion, surge: p.surge });
    const greedy = greedyNearest(s), nr = bestResponse(s, greedy), opt = coordinated(s);
    return { s, greedy, nash: nr.poolOf, opt, rounds: nr.rounds, converged: nr.converged,
      mG: evaluate(s, greedy), mN: evaluate(s, nr.poolOf), mO: evaluate(s, opt) };
  }
  const r = (v, d) => Number(Number(v).toFixed(d));
  const method = (name, m, opt) => ({ method: name, totalCost: r(m.total, 1), meanWait: r(m.meanWait, 1), maxWait: r(m.maxWait, 1),
    meanWaitRed: r(m.meanRed, 1), served: m.served, diverted: m.diverted, divertedRed: m.divertedRed, ratioToOptimal: r(opt > 0 ? m.total / opt : 1, 3) });
  function stats(v) {
    if (!v.length) return { n: 0 };
    v = v.slice().sort((a, b) => a - b);
    const sum = v.reduce((x, y) => x + y, 0), p95 = Math.min(v.length - 1, Math.floor(v.length * 0.95));
    return { n: v.length, mean: r(sum / v.length, 3), median: r(v[Math.floor(v.length / 2)], 3), p95: r(v[p95], 3), max: r(v[v.length - 1], 3) };
  }

  // params: {patients, congestion, surge, seed, trials}
  function runMci(params) {
    const p = Object.assign({ patients: 20, congestion: 1.5, surge: 6, seed: 42, trials: 0 }, params);
    const city = RL.sampleCity();
    const rng = new MT19937(p.seed);
    const first = runOne(city.graph, city.hospitals, rng, p);
    const s = first.s;
    const cnt = { RED: 0, YELLOW: 0, GREEN: 0 };
    s.patients.forEach((e) => cnt[e.priority]++);
    const out = { patients: s.patients.length, red: cnt.RED, yellow: cnt.YELLOW, green: cnt.GREEN, epicentre: s.patients.length ? s.patients[0].place : "",
      methods: [method("Greedy nearest", first.mG, first.mO.total), method("Nash equilibrium", first.mN, first.mO.total), method("Coordinated (MCMF)", first.mO, first.mO.total)],
      pools: s.pools.map((q, i) => ({ pool: q.label, capacity: q.cap, greedy: first.mG.load[i], nash: first.mN.load[i], optimal: first.mO.load[i] })),
      nashRounds: first.rounds, nashConverged: first.converged };
    if (p.trials > 0) {
      const tG = [], tN = [], wG = [], wN = [];
      let notConv = 0, optWorse = 0;
      for (let t = 0; t < p.trials; t++) {
        const x = runOne(city.graph, city.hospitals, rng, p);
        if (!x.converged) notConv++;
        if (x.mO.total > x.mN.total + 1e-3 || x.mO.total > x.mG.total + 1e-3) optWorse++;
        tG.push(x.mG.total / x.mO.total); tN.push(x.mN.total / x.mO.total);
        if (x.mO.waitSum > 0) { wG.push(x.mG.waitSum / x.mO.waitSum); wN.push(x.mN.waitSum / x.mO.waitSum); }
      }
      out.trials = { count: p.trials, totalCostGreedy: stats(tG), totalCostNash: stats(tN), waitGreedy: stats(wG), waitNash: stats(wN),
        nashNotConverged: notConv, optimalWorseThanHeuristic: optWorse };
    }
    return out;
  }

  RL.runMci = runMci;
  RL.MT19937 = MT19937;
})((globalThis.RL = globalThis.RL || {}));

/* =============================================================
   localBackend.js  --  The in-browser stand-in for the C++ server.

   It answers the same requests as engine/include/Api.h (same paths, same status codes, same
   validation messages, same JSON) and delivers the same events the WebSocket would.
   The UI talks to ANY backend through this tiny interface:

       backend.mode                      "local" | "http"
       backend.request(method, path, body) -> Promise<{status, data}>
       backend.onEvent(fn)               fn({type, id, t, data})
       backend.start() / backend.stop()
   ============================================================= */
(function (RL) {
  "use strict";

  const isInt = (v) => Number.isFinite(v) && v === Math.floor(v);
  const TYPES = ["GENERAL", "CARDIAC", "TRAUMA", "NEURO", "BURN", "PEDIATRIC"];

  function createLocalBackend(opts) {
    opts = Object.assign({ tickMs: 200 }, opts);
    const engine = new RL.LocalEngine(opts.engine);
    const listeners = [];
    let timer = null, lastReal = 0;
    const sim = { speed: engine.cfg.simSpeed, paused: false };

    function flush() {
      const ev = engine.drain();
      for (const e of ev) for (const fn of listeners) { try { fn(e); } catch (err) { console.error(err); } }
    }
    const ok = (data, status = 200) => ({ status, data });
    const err = (status, code, message) => ({ status, data: { error: code, message } });
    const validation = (fields) => ({ status: 422, data: { error: "validation", message: "Some fields are missing or invalid.", fields } });
    const fail = (msg) => {
      if (msg.startsWith("unknown")) return err(404, "not_found", msg);
      if (msg.includes("outside the service area")) return validation({ location: msg });
      return err(409, "conflict", msg);
    };
    const pid = (s) => (/^\d{1,9}$/.test(s) ? Number(s) : null);

    function createEmergency(b) {
      const f = {};
      const type = typeof b.type === "string" ? b.type.toUpperCase() : null;
      if (typeof b.type !== "string") f.type = "required";
      else if (!TYPES.includes(type)) f.type = "must be one of GENERAL, CARDIAC, TRAUMA, NEURO, BURN, PEDIATRIC";
      const num = (k) => (typeof b[k] === "number" && Number.isFinite(b[k]) ? b[k] : null);
      const lat = num("lat"), lon = num("lon"), rr = num("respiratoryRate"), pulse = num("pulse");
      if (lat === null) f.lat = "required number"; else if (lat < -90 || lat > 90) f.lat = "must be between -90 and 90";
      if (lon === null) f.lon = "required number"; else if (lon < -180 || lon > 180) f.lon = "must be between -180 and 180";
      if (rr === null) f.respiratoryRate = "required number (breaths per minute)";
      else if (!isInt(rr) || rr < 0 || rr > 80) f.respiratoryRate = "must be a whole number from 0 to 80";
      if (pulse === null) f.pulse = "required number (beats per minute)";
      else if (!isInt(pulse) || pulse < 0 || pulse > 250) f.pulse = "must be a whole number from 0 to 250";
      const avpu = typeof b.avpu === "string" ? b.avpu.toUpperCase() : null;
      if (typeof b.avpu !== "string") f.avpu = "required: A, V, P or U";
      else if (avpu.length !== 1 || !"AVPU".includes(avpu)) f.avpu = "must be A, V, P or U";
      if (typeof b.canWalk !== "boolean") f.canWalk = "required: true or false";
      if (typeof b.severeBleeding !== "boolean") f.severeBleeding = "required: true or false";
      let place = "Reported location";
      if (b.place !== undefined) {
        if (typeof b.place !== "string" || b.place.length > 120) f.place = "must be text up to 120 characters";
        else if (b.place !== "") place = b.place;
      }
      let key = "";
      if (b.idempotencyKey !== undefined) {
        if (typeof b.idempotencyKey !== "string" || b.idempotencyKey.length > 64) f.idempotencyKey = "must be text up to 64 characters";
        else key = b.idempotencyKey;
      }
      if (Object.keys(f).length) return validation(f);
      if (!engine.inServiceArea({ lat, lon })) return validation({ location: "outside the service area" });
      const r = engine.submit(type, { lat, lon }, place, { rr, pulse, avpu, canWalk: b.canWalk, bleed: b.severeBleeding }, key);
      return ok(engine.emergencyJson(r.id), r.duplicate ? 200 : 201);
    }

    function route(method, rawPath, body) {
      const path = rawPath.split("?")[0];
      const p = path.split("/").filter(Boolean);
      body = body || {};
      const GET = method === "GET", POST = method === "POST";
      if (p[0] !== "api" || p.length < 2) return err(404, "not_found", "unknown path");
      if (!GET && !POST) return err(405, "method_not_allowed", "use GET or POST");
      const res = p[1];
      const only = (m) => (m ? null : err(405, "method_not_allowed", "use " + (GET ? "POST" : "GET")));

      if (res === "health" && p.length === 2) return GET ? ok({ status: "ok" }) : only(false);
      if (res === "snapshot" && p.length === 2) return GET ? ok(engine.snapshotJson()) : only(false);
      if (res === "graph" && p.length === 2) return GET ? ok(engine.graphJson()) : only(false);
      if (res === "queue" && p.length === 2) return GET ? ok(engine.queueJson()) : only(false);

      if (res === "emergencies") {
        if (p.length === 2) return GET ? ok(engine.emergenciesJson()) : createEmergency(body);
        const id = pid(p[2]);
        if (id === null) return err(400, "bad_id", "emergency id must be a number");
        if (p.length === 3) {
          if (!GET) return only(false);
          const j = engine.emergencyJson(id);
          return j ? ok(j) : err(404, "not_found", "unknown emergency");
        }
        if (p.length === 4 && POST && (p[3] === "cancel" || p[3] === "discharge")) {
          const m = p[3] === "cancel" ? engine.cancel(id) : engine.discharge(id);
          return m ? fail(m) : ok(engine.emergencyJson(id));
        }
      }
      if (res === "ambulances") {
        if (p.length === 2) return GET ? ok(engine.ambulancesJson()) : only(false);
        const id = pid(p[2]);
        if (id === null) return err(400, "bad_id", "ambulance id must be a number");
        const a = engine.amb(id);
        if (p.length === 3) { if (!GET) return only(false); return a ? ok(engine.ambulanceJson(a)) : err(404, "not_found", "unknown ambulance"); }
        if (p.length === 4 && POST && p[3] === "position") {
          const f = {};
          if (typeof body.lat !== "number" || body.lat < -90 || body.lat > 90) f.lat = "required number between -90 and 90";
          if (typeof body.lon !== "number" || body.lon < -180 || body.lon > 180) f.lon = "required number between -180 and 180";
          if (Object.keys(f).length) return validation(f);
          const m = engine.positionUpdate(id, { lat: body.lat, lon: body.lon });
          return m ? fail(m) : ok(engine.ambulanceJson(engine.amb(id)));
        }
        if (p.length === 4 && POST && p[3] === "maintenance") {
          if (typeof body.on !== "boolean") return validation({ on: "required: true or false" });
          const m = engine.setMaintenance(id, body.on);
          return m ? fail(m) : ok(engine.ambulanceJson(engine.amb(id)));
        }
      }
      if (res === "hospitals") {
        if (p.length === 2) return GET ? ok(engine.hospitalsJson()) : only(false);
        const id = pid(p[2]);
        if (id === null) return err(400, "bad_id", "hospital id must be a number");
        const h = engine.hospital(id);
        if (p.length === 3) { if (!GET) return only(false); return h ? ok(engine.hospitalJson(h)) : err(404, "not_found", "unknown hospital"); }
        if (p.length === 4 && POST) {
          if (p[3] === "beds") {
            const f = {};
            if (!isInt(body.general) || body.general < 0 || body.general > 10000) f.general = "required whole number 0..10000";
            if (!isInt(body.icu) || body.icu < 0 || body.icu > 10000) f.icu = "required whole number 0..10000";
            if (Object.keys(f).length) return validation(f);
            const m = engine.updateBeds(id, body.general, body.icu);
            return m ? fail(m) : ok(engine.hospitalJson(engine.hospital(id)));
          }
          if (p[3] === "heartbeat") { const m = engine.heartbeat(id); return m ? fail(m) : ok(engine.hospitalJson(engine.hospital(id))); }
          if (p[3] === "silence") {
            if (typeof body.silenced !== "boolean") return validation({ silenced: "required: true or false" });
            const m = engine.setHeartbeatEnabled(id, !body.silenced);
            return m ? fail(m) : ok(engine.hospitalJson(engine.hospital(id)));
          }
        }
      }
      if (res === "mci" && p.length === 3 && p[2] === "run") {
        if (!POST) return only(false);
        const f = {}, mp = {};
        const chk = (k, lo, hi, whole, label) => {
          if (body[k] === undefined) return;
          if (typeof body[k] !== "number" || !Number.isFinite(body[k]) || (whole && !isInt(body[k])) || body[k] < lo || body[k] > hi) f[k] = label;
          else mp[k] = body[k];
        };
        chk("patients", 2, 60, true, "whole number 2..60");
        chk("congestion", 0, 50, false, "number 0..50");
        chk("surge", 1, 30, true, "whole number 1..30");
        chk("seed", 0, 4294967295, true, "whole number 0..4294967295");
        chk("trials", 0, 500, true, "whole number 0..500");
        if (Object.keys(f).length) return validation(f);
        return ok(RL.runMci(mp));
      }
      return err(404, "not_found", "unknown path");
    }

    const backend = {
      mode: "local",
      setAuth() { /* the in-browser preview has no accounts */ },
      engine, sim,
      request(method, path, body) {
        let out;
        try { out = route(method, path, body); } catch (e) { out = err(500, "internal", String(e.message || e)); }
        flush();
        return Promise.resolve(JSON.parse(JSON.stringify(out)));   // same copy semantics as a network round trip
      },
      onEvent(fn) { listeners.push(fn); },
      start() {
        engine.emitInitialState(); flush();
        lastReal = performance.now();
        timer = setInterval(() => {
          const now = performance.now();
          const dt = Math.min(1.0, (now - lastReal) / 1000);
          lastReal = now;
          if (!sim.paused) { engine.tick(dt); flush(); }
        }, opts.tickMs);
        listeners.forEach((fn) => fn({ type: "connection", id: 0, t: 0, data: { connected: true } }));
      },
      stop() { if (timer) clearInterval(timer); timer = null; },
      // local-only controls (the real server reads these from its environment)
      setSpeed(x) { sim.speed = x; engine.cfg.simSpeed = x; },
      setPaused(p) { sim.paused = p; },
      // test helper: advance without waiting
      advance(seconds, step = 0.5) { for (let t = 0; t < seconds; t += step) engine.tick(step); flush(); },
    };
    return backend;
  }

  RL.createLocalBackend = createLocalBackend;
})((globalThis.RL = globalThis.RL || {}));
